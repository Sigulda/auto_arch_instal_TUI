##### нужно обновить
создание сервиса разового запуска tailscale для авторизации в мою меш сеть
```
# Создаем сервис для автоматической авторизации при первом запуске
cat > /etc/systemd/system/tailscale-setup.service <<EOL
[Unit]
Description=Authenticate Tailscale on first boot
After=network-online.target tailscaled.service
Wants=network-online.target

[Service]
Type=oneshot
# Вставь свой ключ ниже вместо TS_AUTH_KEY
ExecStart=/usr/bin/tailscale up --authkey=tskey-auth-kXXXXX-XXXXXXXXXXXXXXXX
# Самоликвидация сервиса после успеха
ExecStartPost=/usr/bin/systemctl disable tailscale-setup.service

[Install]
WantedBy=multi-user.target
EOL

# Включаем этот временный сервис внутри postinstall.sh
systemctl enable tailscale-setup.service

``` 
можно прописать в одну строку внутри postinstall.sh или создать отдельным файлом
